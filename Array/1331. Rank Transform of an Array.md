# 1331. Rank Transform of an Array
difficulty: `easy`
tags: #Senior #Array #Greedy #WeeklyContest468

### Problem
Given an array of integers arr, replace each element with its rank.
The rank represents how large the element is. The rank has the following rules:
Rank is an integer starting from 1.
The larger the element, the larger the rank. If two elements are equal, their rank must be the same.
Rank should be as small as possible.

```
Example 1:
Input: arr = [37,12,28,9,100,56,80,5,12]
Output: [5,3,4,2,8,6,7,1,3]
```

### Solution 1
```c =
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* arrayRankTransform(int* arr, int arrSize, int* returnSize) {
    int* result = (int*)malloc(sizeof(int) * arrSize);
    * returnSize = arrSize;
    int min = INT_MAX, count = arrSize, rank = 1;
    
    while (count){
        //find min
        min = INT_MAX;
        for (int i = 0 ; i < arrSize; i++){
            if(min > arr[i]) min = arr[i];
        }

        for (int i = 0 ; i < arrSize; i++){
            if(min == arr[i]) {
                arr[i] = INT_MAX;
                result[i] = rank;
                count--;
            }
        }
        rank++;
    }

    return result;
}
```

### Solution 2
```c
/* ---------- comparator: compare instead of subtract, so ±1e9 can't overflow ---------- */
static int cmpInt(const void* a, const void* b) {
    int x = *(const int*)a, y = *(const int*)b;
    return (x > y) - (x < y);
}

/* ---------- helpers: a prime table size spreads keys better than a round number ---------- */
static bool isPrime(int x) {
    if (x < 2) return false;
    for (long i = 2; i * i <= x; i++) if (x % i == 0) return false;
    return true;
}
static int nextPrime(int x) {
    while (!isPrime(x)) x++;
    return x;
}

int* arrayRankTransform(int* arr, int arrSize, int* returnSize) {
    *returnSize = arrSize;
    if (arrSize == 0) return NULL;

    /* ================= SORTING: establish the rank order ================= */

    /* Work on a copy so the caller's array keeps its original order */
    int* sorted = (int*)malloc(sizeof(int) * arrSize);
    for (int i = 0; i < arrSize; i++) sorted[i] = arr[i];
    qsort(sorted, arrSize, sizeof(int), cmpInt);

    /* Squeeze out duplicates in place. m = number of distinct values.
       Safe because the write index m never runs ahead of the read index i. */
    int m = 0;
    for (int i = 0; i < arrSize; i++) {
        if (i == 0 || sorted[i] != sorted[i - 1]) sorted[m++] = sorted[i];
    }
    /* After this, sorted[0..m-1] is ascending and distinct,
       so a value sitting at index p has rank p+1. */

    /* ================= HASH TABLE: value -> rank, O(1) lookup ================= */

    /* Load factor ~0.5 keeps linear probing short and guarantees a free slot */
    int tableSize = nextPrime(m * 2 + 1);
    int*  keys = (int*)malloc(sizeof(int) * tableSize);
    int*  vals = (int*)malloc(sizeof(int) * tableSize);
    bool* used = (bool*)calloc(tableSize, sizeof(bool));

    for (int i = 0; i < m; i++) {
        /* Knuth multiplicative hashing: the multiply scrambles the bits so that
           structured inputs (arithmetic sequences, multiples of tableSize) don't
           all collide into one bucket. */
        unsigned int h = ((unsigned int)sorted[i]) * 2654435761u % tableSize;
        while (used[h]) h = (h + 1) % tableSize;   /* linear probing */
        keys[h] = sorted[i];
        vals[h] = i + 1;                            /* ranks start at 1 */
        used[h] = true;
    }

    /* ================= ARRAY: write answers back in the original order ======= */

    int* result = (int*)malloc(sizeof(int) * arrSize);
    for (int i = 0; i < arrSize; i++) {
        unsigned int h = ((unsigned int)arr[i]) * 2654435761u % tableSize;
        /* Every value was inserted above, so this always terminates on a hit */
        while (!(used[h] && keys[h] == arr[i])) h = (h + 1) % tableSize;
        result[i] = vals[h];
    }

    free(sorted);
    free(keys);
    free(vals);
    free(used);
    return result;
}
```
